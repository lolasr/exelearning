Plantilla universal de Prompt

Esqueleto del Prompt

Quiero que generes [tipo de recurso] sobre el tema de [tema] para alumnado de [nivel]. El recurso debe servir para [objetivo]. Quiero que tenga [elementos específicos]. Hazlo con estilo [estilo deseado, si lo hay].


Ejemplo Práctico

“Quiero que generes un juego interactivo sobre el tema de trigonometría para alumnado de 3º de ESO. El recurso debe servir para repasar para el examen. Quiero que tenga preguntas tipo test, retroalimentación automática y un contador de puntos. Hazlo con estilo librojuego de aventura.”



A continuación, prueba a copiar el codigo siguiente en un fichero de nota y a guardarlo en tu equipo como UTF-8 y con extensión .html, podrás abrirlo con cualquier navegador y comenzar a usar el recurso generado.

<script>
/*
  Trigonaventura - Juego librojuego de trigonometría
  Autor: generado para uso educativo
  Estructura: capítulos. Cada capítulo contiene preguntas (multiple-choice).
  Reglas: al superar el umbral (70%) del capítulo, desbloqueas el siguiente capítulo y una recompensa.
*/

// Datos del juego (3 capítulos como ejemplo)
const chapters = [
  {
    title: "Capítulo 1 — El Valle de los Triángulos",
    story: "Eres un explorador que llega al Valle de los Triángulos. Para encontrar la primera llave debes resolver problemas básicos sobre triángulos rectángulos y razones trigonométricas.",
    thresholdPercent: 70,
    reward: {name:"Llave de Madera",desc:"Permite abrir cofres pequeños."},
    questions: [
      {
        q:"En un triángulo rectángulo, ¿qué representa el seno de un ángulo agudo?",
        choices:["Cateto adyacente dividido entre hipotenusa","Cateto opuesto dividido entre hipotenusa","Cateto opuesto dividido entre cateto adyacente","Hipotenusa dividida entre cateto opuesto"],
        answer:1,
        explanation:"Por definición, seno(θ) = cateto opuesto / hipotenusa."
      },
      {
        q:"Calcula sin(30°). Elige la respuesta correcta.",
        choices:["1","√3/2","1/2","√2/2"],
        answer:2,
        explanation:"sin(30°) = 1/2."
      },
      {
        q:"En un triángulo rectángulo con hipotenusa 10 y cateto opuesto 6, ¿cuál es el valor aproximado de sin(θ)? (donde θ está frente al cateto opuesto)",
        choices:["0.6","0.8","1.6","0.4"],
        answer:0,
        explanation:"sin(θ) = opuesto/hipotenusa = 6/10 = 0.6."
      }
    ]
  },
  {
    title: "Capítulo 2 — La Cueva de los Ángulos",
    story: "Dentro de la cueva aparecen trampas que solo se desactivan con conocimiento de ángulos especiales y relaciones entre seno, coseno y tangente.",
    thresholdPercent: 70,
    reward: {name:"Mapa Parcial",desc:"Muestra la zona noreste del mapa."},
    questions: [
      {
        q:"¿Cuál es el coseno de 60°?",
        choices:["1/2","√3/2","√2/2","0"],
        answer:0,
        explanation:"cos(60°) = 1/2."
      },
      {
        q:"Si tan(θ)=1, ¿cuál es el ángulo agudo θ (grados)?",
        choices:["30°","45°","60°","90°"],
        answer:1,
        explanation:"tan(45°)=1."
      },
      {
        q:"En un triángulo rectángulo, si cos(θ)=0.8 y la hipotenusa mide 5, ¿cuál es el cateto adyacente?",
        choices:["3","4","2","5"],
        answer:1,
        explanation:"cos(θ)=adyacente/hipotenusa → adyacente=0.8×5=4."
      }
    ]
  },
  {
    title: "Capítulo 3 — La Torre del Teorema",
    story: "Has llegado a la torre final. Para obtener el tesoro debes aplicar trigonometría en problemas y comprender explicaciones paso a paso.",
    thresholdPercent: 70,
    reward: {name:"Cristal del Saber",desc:"Revela las soluciones correctas al practicar."},
    questions: [
      {
        q:"En un triángulo rectángulo, ¿qué relación da la tangente?",
        choices:["opuesto/hipotenusa","adyacente/hipotenusa","opuesto/adyacente","hipotenusa/adyacente"],
        answer:2,
        explanation:"tan(θ) = cateto opuesto / cateto adyacente."
      },
      {
        q:"Calcula tan(45°).",
        choices:["0","1","√3","undefined"],
        answer:1,
        explanation:"tan(45°) = 1."
      },
      {
        q:"Problema aplicado: Un árbol proyecta una sombra de 8 m cuando el sol forma un ángulo de 30° con el suelo. ¿Cuál es la altura aproximada del árbol? (usa tan(30°)=1/√3≈0.577)",
        choices:["≈4.6 m","≈13.8 m","≈8.0 m","≈6.9 m"],
        answer:0,
        explanation:"tan(30°)=altura/8 → altura=8×tan(30°)≈8×0.577≈4.616 m."
      }
    ]
  }
];

// Estado del juego
let state = {
  chapterIndex: 0,
  totalPoints: 0,
  chapterPoints: 0,
  completedChapters: [],
  rewards: []
};

const chapterArea = document.getElementById('chapterArea');
const chapterIndexEl = document.getElementById('chapterIndex');
const chapterCountEl = document.getElementById('chapterCount');
const totalPointsEl = document.getElementById('totalPoints');
const chapterScoreEl = document.getElementById('chapterScore');
const progressBar = document.getElementById('progressBar');
const rewardsList = document.getElementById('rewardsList');
const prevBtn = document.getElementById('prevBtn');
const nextBtn = document.getElementById('nextBtn');

chapterCountEl.textContent = chapters.length;

function renderChapter(i){
  const ch = chapters[i];
  chapterArea.innerHTML = '';
  chapterIndexEl.textContent = i+1;
  // Story
  const storyDiv = document.createElement('div');
  storyDiv.className = 'story';
  storyDiv.innerHTML = `<h2 style="margin-top:0">${ch.title}</h2><p>${ch.story}</p>`;
  chapterArea.appendChild(storyDiv);

  // Questions form
  const form = document.createElement('form');
  form.id = 'quizForm';
  ch.questions.forEach((qs, idx) => {
    const qDiv = document.createElement('div');
    qDiv.className = 'question';
    qDiv.innerHTML = `<div style="font-weight:700">Pregunta ${idx+1}.</div><div style="margin-top:6px">${qs.q}</div>`;
    const choicesDiv = document.createElement('div');
    choicesDiv.className = 'choices';
    qs.choices.forEach((c, ci) => {
      const id = `q${idx}_c${ci}`;
      const label = document.createElement('label');
      label.innerHTML = `<input type="radio" name="q${idx}" value="${ci}" id="${id}"> ${c}`;
      choicesDiv.appendChild(label);
    });
    qDiv.appendChild(choicesDiv);
    // placeholder for feedback
    const feedback = document.createElement('div');
    feedback.className = 'feedback hidden';
    feedback.id = `feedback-${idx}`;
    qDiv.appendChild(feedback);

    form.appendChild(qDiv);
  });

  // Controls: submit + show answers + reset
  const controls = document.createElement('div');
  controls.className = 'controls';
  const submitBtn = document.createElement('button');
  submitBtn.textContent = 'Comprobar respuestas';
  submitBtn.type = 'button';
  submitBtn.onclick = () => checkAnswers(i);
  const showAllBtn = document.createElement('button');
  showAllBtn.textContent = 'Mostrar soluciones';
  showAllBtn.className = 'secondary';
  showAllBtn.type = 'button';
  showAllBtn.onclick = () => revealAll(i);
  const resetBtn = document.createElement('button');
  resetBtn.textContent = 'Reiniciar capítulo';
  resetBtn.className = 'secondary';
  resetBtn.type = 'button';
  resetBtn.onclick = () => { if(confirm('¿Reiniciar las respuestas de este capítulo?')) renderChapter(i); };

  controls.append(submitBtn, showAllBtn, resetBtn);
  chapterArea.appendChild(form);
  chapterArea.appendChild(controls);

  // Update sidebar info
  updateSidebar(0, ch.questions.length);
  // Prev/next buttons enable state
  prevBtn.disabled = (i===0);
  nextBtn.disabled = (i >= chapters.length-1) && !state.completedChapters.includes(i);
}

function checkAnswers(chIndex){
  const ch = chapters[chIndex];
  const form = document.getElementById('quizForm');
  let correctCount = 0;
  ch.questions.forEach((q, idx) => {
    const selected = form.querySelector(`input[name="q${idx}"]:checked`);
    const feedbackDiv = document.getElementById(`feedback-${idx}`);
    feedbackDiv.className = 'feedback';
    if(!selected){
      feedbackDiv.classList.add('wrong');
      feedbackDiv.innerHTML = `<strong>Sin respuesta.</strong> ${q.explanation} <div style="margin-top:8px"><small>Selecciona una opción y vuelve a comprobar.</small></div>`;
    } else {
      const val = parseInt(selected.value,10);
      if(val === q.answer){
        correctCount++;
        feedbackDiv.classList.add('ok');
        feedbackDiv.innerHTML = `<strong>Correcto.</strong> ${q.explanation}`;
      } else {
        feedbackDiv.classList.add('wrong');
        const correctText = q.choices[q.answer];
        feedbackDiv.innerHTML = `<strong>Incorrecto.</strong> La respuesta correcta es: <em>${correctText}</em>. ${q.explanation}`;
      }
    }
  });

  // Puntos por capítulo: 10 puntos por pregunta correcta
  const pointsGained = correctCount * 10;
  // Update state: For simplicity, chapterPoints holds current chapter points (not cumulative)
  state.chapterPoints = pointsGained;
  // Update total points (only add if first time passing this chapter or later recalculation allows higher)
  // We'll store best chapter score
  const chapterKey = `chapter_best_${chIndex}`;
  const prevBest = parseInt(localStorage.getItem(chapterKey) || '0',10);
  if(pointsGained > prevBest){
    localStorage.setItem(chapterKey, pointsGained);
  }
  // Compute threshold
  const percent = (correctCount / ch.questions.length) * 100;
  let passed = percent >= ch.thresholdPercent;

  // Award reward if passed and not awarded before
  if(passed && !state.completedChapters.includes(chIndex)){
    state.completedChapters.push(chIndex);
    state.rewards.push(ch.reward);
    // Add points to total
    state.totalPoints += pointsGained;
    alert(`¡Has superado el capítulo! Recompensa: ${ch.reward.name}`);
  } else {
    // If already completed before, total points is sum of bests
    // Recompute totalPoints as sum of bests
    let total = 0;
    for(let i=0;i<chapters.length;i++){
      total += parseInt(localStorage.getItem(`chapter_best_${i}`) || '0',10);
    }
    state.totalPoints = total;
  }

  updateSidebar(pointsGained, ch.questions.length, percent);
  renderRewards();
  saveProgress();
}

function revealAll(chIndex){
  // Show answers and explanations
  const ch = chapters[chIndex];
  ch.questions.forEach((q, idx) => {
    const feedbackDiv = document.getElementById(`feedback-${idx}`);
    feedbackDiv.className = 'feedback';
    const correctText = q.choices[q.answer];
    feedbackDiv.innerHTML = `<strong>Solución:</strong> <em>${correctText}</em>. ${q.explanation}`;
  });
}

function updateSidebar(chapterPoints = 0, totalQs = 1, percent = 0){
  chapterScoreEl.textContent = chapterPoints;
  const progPercent = Math.round((chapterPoints / (totalQs*10))*100);
  progressBar.style.width = progPercent + '%';
  totalPointsEl.textContent = state.totalPoints;
  // Show simple progress text
  // If passed and not in completedChapters, show hint to press Siguiente
  const idx = state.chapterIndex;
  const ch = chapters[idx];
  if(percent >= ch.thresholdPercent && !state.completedChapters.includes(idx)){
    // unlock next button
    nextBtn.disabled = (idx >= chapters.length-1) ? true : false;
  } else {
    // Only allow next if already completed before
    nextBtn.disabled = !state.completedChapters.includes(idx) && (idx >= chapters.length-1);
  }
}

function renderRewards(){
  rewardsList.innerHTML = '';
  if(state.rewards.length===0){ rewardsList.innerHTML = '<div class="meta">Aún no tienes recompensas</div>'; return; }
  state.rewards.forEach(r => {
    const d = document.createElement('div');
    d.className = 'reward';
    d.innerHTML = `<div class="badge">${r.name.split(' ')[0]}</div><div><strong>${r.name}</strong><div style="font-size:13px;color:var(--muted)">${r.desc}</div></div>`;
    rewardsList.appendChild(d);
  });
}

function saveProgress(){
  localStorage.setItem('trigo_totalPoints', state.totalPoints);
  localStorage.setItem('trigo_completed', JSON.stringify(state.completedChapters));
  localStorage.setItem('trigo_rewards', JSON.stringify(state.rewards));
}

function loadProgress(){
  const completed = JSON.parse(localStorage.getItem('trigo_completed') || '[]');
  const rewards = JSON.parse(localStorage.getItem('trigo_rewards') || '[]');
  state.completedChapters = completed;
  state.rewards = rewards;
  // recompute total from stored bests
  let total = 0;
  for(let i=0;i<chapters.length;i++) total += parseInt(localStorage.getItem(`chapter_best_${i}`) || '0',10);
  state.totalPoints = total;
  totalPointsEl.textContent = state.totalPoints;
  renderRewards();
}

// Navigation
prevBtn.addEventListener('click', () => {
  if(state.chapterIndex > 0){
    state.chapterIndex--;
    renderChapter(state.chapterIndex);
  }
});
nextBtn.addEventListener('click', () => {
  const next = state.chapterIndex + 1;
  // allow next only if unlocked
  if(next < chapters.length){
    if(state.completedChapters.includes(state.chapterIndex) || state.completedChapters.includes(next)){
      state.chapterIndex = next;
      renderChapter(state.chapterIndex);
    } else {
      alert('No has desbloqueado aún el siguiente capítulo. Supera el capítulo actual para avanzar.');
    }
  }
});

// Inicialización
loadProgress();
renderChapter(state.chapterIndex);
</script>
</body>
</html>
